# Keep Google Play Games Services + Play Services from R8/ProGuard minification.
#
# This project builds with minification ON for both debug and release
# (ProjectSettings: AndroidMinifyDebug/Release = 1). The Play Games classes are
# invoked from Unity C# through JNI reflection (AndroidJavaClass/AndroidJavaObject),
# so R8 cannot see the usage and strips them — producing at runtime:
#   java.lang.ClassNotFoundException: com.google.android.gms.games.Games
# These keep rules prevent that. Do not remove while minification is enabled.

# Google Play Games Unity plugin native support library
-keep class com.google.games.** { *; }
-keep interface com.google.games.** { *; }

# Google Play Games Services SDK (com.google.android.gms.games.Games, etc.)
-keep class com.google.android.gms.games.** { *; }
-keep interface com.google.android.gms.games.** { *; }

# Supporting Play Services used by the plugin (auth, common, tasks, nearby)
-keep class com.google.android.gms.auth.** { *; }
-keep class com.google.android.gms.common.** { *; }
-keep class com.google.android.gms.tasks.** { *; }
-keep class com.google.android.gms.nearby.** { *; }

# Silence warnings about optional/unreferenced Play Services symbols
-dontwarn com.google.android.gms.**
